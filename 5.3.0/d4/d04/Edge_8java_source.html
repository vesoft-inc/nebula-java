33 public Edge(int graphId,

39 ResultGraphSchemas graphSchemas) {
40 this.graphId = graphId;
41 this.graphName = graphSchemas.getGraphSchema(graphId).getGraphName();
42 this.edgeTypeId = edgeTypeId;
43 int noDirectedTypeId = edgeTypeId & 0x3FFFFFFF;
44 this.edgeTypeName = graphSchemas
45 .getGraphSchema(graphId)
46 .getEdgeSchema(noDirectedTypeId)
47 .getEdgeTypeName();
48 this.labels = graphSchemas
49 .getGraphSchema(graphId)
50 .getEdgeSchema(noDirectedTypeId)
51 .getEdgeLabels();
52 this.rank = rank;
53 this.properties = properties;
54
55 int edgeTypeMoveBits = EDGE_TYPE_ID_SIZE * 8 - 2;
56 int directionBits = (edgeTypeId >> edgeTypeMoveBits) & 0x3;
57 switch (directionBits) {
58 case 0x0:
59 this.direction = Direction.OUTGOING;
60 break;
61 case 0x1:
62 this.direction = Direction.INCOMING;
63 break;
64 case 0x2:
65 case 0x3:
66 this.direction = Direction.UNDIRECTED;
67 break;
68 default:
69 this.direction = Direction.KNOWN;
70 }
71 if (this.direction == Direction.INCOMING) {
72 this.srcId = dstId;
73 this.dstId = srcId;
74 } else {
75 this.srcId = srcId;
76 this.dstId = dstId;
77 }
78 }
