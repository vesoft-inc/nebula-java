26 public static List<HostAddress> validateAddress(String addresses) throws UnknownHostException {
27 List<HostAddress> newAddrs = new ArrayList<>();
28 for (String addr : addresses.split(",")) {
29 if (addr == null || addr.isEmpty()) {
30 continue;
31 }
32 String host;
33 String portString = null;
34
35 if (addr.startsWith("[")) {
36 String[] hostAndPort = getHostAndPortFromBracketedAddr(addr);
37 host = hostAndPort[0];
38 portString = hostAndPort[1].trim();
39 } else {
40 int colonPos = addr.indexOf(":");
41 if (colonPos >= 0 && addr.indexOf(":", colonPos + 1) == -1) {
42 // exactly 1 colon.
43 host = addr.substring(0, colonPos);
44 portString = addr.substring(colonPos + 1).trim();
45 } else if (colonPos == -1) {
46 // no colon
47 throw new IllegalArgumentException("No port:" + addr);
48 } else {
49 // many colons
50 throw new IllegalArgumentException(
51 "Possible bracketless IPv6 literal: " + addr);
52 }
53 }
54
55 int port;
56 try {
57 port = Integer.parseInt(portString);
58 } catch (NumberFormatException e) {
59 throw new IllegalArgumentException("Unparseable port number: " + addr);
60 }
61
62 // get all host name
63 InetAddress[] inetAddresses = InetAddress.getAllByName(host);
64 for (InetAddress inetAddress : inetAddresses) {
65 String ip = inetAddress.getHostAddress();
66 if (!(InetAddresses.isInetAddress(ip)
67 || InetAddresses.isUriInetAddress(ip)
68 || InternetDomainName.isValid(ip))
69 || (port <= 0 || port >= 65535)) {
70 throw new IllegalArgumentException(
71 String.format("host %s and port %d is invalid.", ip, port));
72 }
73 }
74 newAddrs.add(new HostAddress(host, port));
75 }
76 return newAddrs;
77 }

79 private static String[] getHostAndPortFromBracketedAddr(String hostPortString) {
80 int colonIndex = hostPortString.indexOf(':');
81 int closeBracketIndex = hostPortString.lastIndexOf(']');
82 if (colonIndex == -1 || closeBracketIndex <= colonIndex) {
83 throw new IllegalArgumentException("Invalid bracketed host/port: " + hostPortString);
84 }
85 String host = hostPortString.substring(1, closeBracketIndex);
86 if (closeBracketIndex + 1 == hostPortString.length()) {
87 throw new IllegalArgumentException("Invalid port for addr:" + hostPortString);
88 } else {
89 if (hostPortString.charAt(closeBracketIndex + 1) != ':') {
90 throw new IllegalArgumentException(
91 "Only a colon may follow a close bracket:" + hostPortString);
92 }
93 for (int i = closeBracketIndex + 2; i < hostPortString.length(); ++i) {
94 if (!Character.isDigit(hostPortString.charAt(i))) {
95 throw new IllegalArgumentException("Port must be numeric: " + hostPortString);
96 }
97 }
98 return new String[]{host, hostPortString.substring(closeBracketIndex + 2)};
99 }
