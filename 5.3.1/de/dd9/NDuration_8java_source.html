112 durationStr.append("T");

114 if (getHour() != 0) {

119 }
120 if (getSecond() != 0

122 || (getDay() == 0 && getHour() == 0 && getMinute() == 0 && getSecond() == 0)) {
123 if (getMicrosecond() == 0) {
124 durationStr.append(getSecond()).append("S");
125 } else {
126 int totalMicroseconds = getSecond() * 1000000 + getMicrosecond();
127 boolean isMinus = totalMicroseconds < 0;
128 if (isMinus) {
129 totalMicroseconds = -totalMicroseconds;
130 }
131 int seconds = totalMicroseconds / 1000000;
132 int microSec = totalMicroseconds % 1000000;
133 if (isMinus) {
134 durationStr
135 .append(String.format("-%d.%06d", seconds, microSec));
136 } else {
137 durationStr
138 .append(String.format("%d.%06d", seconds, microSec));
139 }
140 while (durationStr.charAt(durationStr.length() - 1) == '0') {
141 durationStr.setLength(durationStr.length() - 1);
142 }
143 durationStr.append("S");
144 }
