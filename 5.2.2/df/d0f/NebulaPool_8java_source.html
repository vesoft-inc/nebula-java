52 private NebulaPool(Builder builder) throws IOErrorException, AuthFailedException {
53 if (hasInit.get()) {
54 return;
55 }
56 this.maxWaitMills = builder.maxWaitMills;
57 this.maxLifeMills = builder.maxLifeTimeMs;
58 this.loadBalancer = new RoundRobinLoadBalancer(builder);
59
60 loadBalancer.checkServers();
61 // pool config
62 GenericObjectPoolConfig objConfig = new GenericObjectPoolConfig();
63 objConfig.setMaxIdle(builder.maxClientSize);
64 objConfig.setMinIdle(builder.minClientSize);
65 objConfig.setMaxTotal(builder.maxClientSize);
66 objConfig.setBlockWhenExhausted(builder.blockWhenExhausted);
67 objConfig.setMaxWaitMillis(builder.maxWaitMills);
68 objConfig.setTimeBetweenEvictionRunsMillis(builder.idleEvictScheduleMills);
69 objConfig.setMinEvictableIdleTimeMillis(builder.minEvictableIdleTimeMillis);
70 objConfig.setTestOnBorrow(builder.testOnBorrow);
71 objConfig.setLifo(builder.lifo);
72 // just test the validation when session is idle.
73 if (builder.healthCheckTimeMills > 0) {
74 objConfig.setTestWhileIdle(true);
75 objConfig.setTimeBetweenEvictionRunsMillis(builder.healthCheckTimeMills);
76 }
77
78 ClientPoolFactory factory = new ClientPoolFactory(
79 loadBalancer,
80 builder);
81 pool = new GenericObjectPool<>(factory, objConfig);
82 hasInit.compareAndSet(false, true);
83 }

98 public void returnClient(NebulaClient client) {
99 if (client.isClosed()
100 || (System.currentTimeMillis() - client.getCreateTime()) >= maxLifeMills) {
101 try {
102 pool.invalidateObject(client);
103 } catch (Exception e) {
104 throw new RuntimeException(e);
105 }
106 } else {
107 pool.returnObject(client);
108 }
109 }
