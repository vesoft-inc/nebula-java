8public class ZoneOffsetUtil {
9 static final int SECONDS_PER_MINUTE = 60;
10 static final int MINUTES_PER_HOUR = 60;
11 static final int SECONDS_PER_HOUR = SECONDS_PER_MINUTE * MINUTES_PER_HOUR;
12
19 public static String buildOffset(int offsetSeconds) {
20 if (offsetSeconds == 0) {
21 return "Z";
22 } else {
23 int absOffsetSeconds = Math.abs(offsetSeconds);
24 StringBuilder buf = new StringBuilder();
25 int absHours = absOffsetSeconds / SECONDS_PER_HOUR;
26 int absMinutes = (absOffsetSeconds / SECONDS_PER_MINUTE) % MINUTES_PER_HOUR;
27 buf.append(offsetSeconds < 0 ? "-" : "+")
28 .append(absHours < 10 ? "0" : "").append(absHours)
29 .append(absMinutes < 10 ? ":0" : ":").append(absMinutes);
30 int absSeconds = absOffsetSeconds % SECONDS_PER_MINUTE;
31 if (absSeconds != 0) {
32 buf.append(absSeconds < 10 ? ":0" : ":").append(absSeconds);
33 }
34 return buf.toString();
35 }
36 }
