51 private NebulaPool(Builder builder) throws IOErrorException, AuthFailedException {
52 if (hasInit.get()) {
53 return;
54 }
55 this.maxWaitMills = builder.maxWaitMills;
56 this.maxLifeMills = builder.maxLifeTimeMs;
57 this.loadBalancer = new RoundRobinLoadBalancer(builder);
58
59 loadBalancer.checkServers();
60 // pool config
61 GenericObjectPoolConfig objConfig = new GenericObjectPoolConfig();
62 objConfig.setMaxIdle(builder.maxClientSize);
63 objConfig.setMinIdle(builder.minClientSize);
64 objConfig.setMaxTotal(builder.maxClientSize);
65 objConfig.setBlockWhenExhausted(builder.blockWhenExhausted);
66 objConfig.setMaxWaitMillis(builder.maxWaitMills);
67 objConfig.setTimeBetweenEvictionRunsMillis(builder.idleEvictScheduleMills);
68 objConfig.setMinEvictableIdleTimeMillis(builder.minEvictableIdleTimeMillis);
69 objConfig.setTestOnBorrow(builder.testOnBorrow);
70 objConfig.setLifo(builder.lifo);
71 // just test the validation when session is idle.
72 if (builder.healthCheckTimeMills > 0) {
73 objConfig.setTestWhileIdle(true);
74 objConfig.setTimeBetweenEvictionRunsMillis(builder.healthCheckTimeMills);
75 }
76
77 ClientPoolFactory factory = new ClientPoolFactory(
78 loadBalancer,
79 builder);
80 pool = new GenericObjectPool<>(factory, objConfig);
81 hasInit.compareAndSet(false, true);
82 }

97 public void returnClient(NebulaClient client) {
98 if (client.isClosed()
99 || (System.currentTimeMillis() - client.getCreateTime()) >= maxLifeMills) {
100 try {
101 pool.invalidateObject(client);
102 } catch (Exception e) {
103 throw new RuntimeException(e);
104 }
105 } else {
106 pool.returnObject(client);
107 }
108 }
