192 private void initClient() throws AuthFailedException, IOErrorException {
193 // create connection with NebulaGraph Server
194 AuthResult authResult = null;
195 connection = new GrpcConnection();
196 int tryConnectTimes = servers.size();
197 Collections.shuffle(servers);
198 while (tryConnectTimes-- > 0) {
199 try {
200 connection.open(servers.get(tryConnectTimes), builder);
201 authResult = connection.authenticate(builder.userName, builder.authOptions);
202 sessionId = authResult.getSessionId();
203 version = authResult.getVersion();
204 createTime = System.currentTimeMillis();
205 break;
206 } catch (AuthFailedException e) {
207 logger.error("create NebulaClient failed.", e);
208 throw e;
209 } catch (Exception e) {
210 if (tryConnectTimes == 0) {
211 logger.error("create NebulaClient failed.", e);
212 throw e;
213 }
214 }
215 }
216 }

388 private ScanNodeResultIterator scanNode(String schema,
389 String graphName,
390 String nodeType,
391 List<String> returnProperties,
392 boolean allProperties,
393 List<Integer> parts,
394 int batchSize) {
395 // get node type's all property names
396 List<String> nodeProperties = null;
397 try {
398 nodeProperties = getNodeProperties(graphName, nodeType);
399 } catch (IOErrorException e) {
400 logger.error("get node schema failed.", e);
401 throw new RuntimeException(e);
402 }
403
404 // construct the return property list for scan
405 List<String> propertyList = new ArrayList<>();
406 if (allProperties) {
407 propertyList.addAll(nodeProperties);
408 } else {
409 String primaryKey = nodeProperties.get(0);
410 // put the primary key always on the head of the propertyList for scan
411 propertyList.add(primaryKey);
412 for (String propName : returnProperties) {
413 if (propName.trim().equals(primaryKey)) {
414 continue;
415 }
416 propertyList.add(propName);
417 }
418 }
419
420 return new ScanNodeResultIterator(schema,
421 graphName,
422 nodeType,
423 propertyList,
424 parts,
425 batchSize,
426 builder.scanParallel,
427 servers,
428 builder);
429 }

658 private List<Integer> getAllParts() {
659 String showPartitions = "CALL show_partitions() RETURN *";
660 ResultSet resultSet;
661 try {
662 resultSet = execute(showPartitions);
663 } catch (Exception e) {
664 logger.error("get all partitions error", e);
665 throw new RuntimeException("get all partitions error", e);
666 }
667 if (!resultSet.isSucceeded() || resultSet.isEmpty()) {
668 logger.error("get all partitions failed for {}", resultSet.getErrorMessage());
669 throw new RuntimeException(
670 "get all partitions failed for " + resultSet.getErrorMessage());
671 }
672
673 List<Integer> partitions = new ArrayList<>();
674 while (resultSet.hasNext()) {
675 partitions.add(resultSet.next().get("partition_id").asInt());
676 }
677 if (partitions.contains(0)) {
678 partitions.remove(Integer.valueOf(0));
679 }
680 return partitions;
681 }

691 private List<String> getNodeProperties(String graphName, String nodeType)
692 throws IOErrorException {
693 String graphType = getGraphType(graphName);
694 String descNodeType = String.format("DESCRIBE NODE TYPE `%s` OF `%s`",
695 GqlUtil.escape(nodeType),
696 GqlUtil.escape(graphType));
697 ResultSet resultSet = execute(descNodeType);
698 if (!resultSet.isSucceeded() || resultSet.isEmpty()) {
699 logger.error(String.format("get description of %s failed for %s", nodeType,
700 resultSet.getErrorMessage()));
701 throw new IllegalArgumentException(String.format("node type %s does not exist in %s",
702 nodeType, graphName));
703 }
704
705 List<String> pks = new ArrayList<>();
706 List<String> propNames = new ArrayList<>();
707 while (resultSet.hasNext()) {
708 ResultSet.Record record = resultSet.next();
709 propNames.add(record.get("property_name").asString());
710 if ("Y".equals(record.get("primary_key").asString())) {
711 pks.add(record.get("property_name").asString());
712 }
713 }
714
715 if (pks.isEmpty()) {
716 logger.error("node type " + nodeType + " has no primary key.");
717 throw new RuntimeException("node type " + nodeType + " has no primary key");
718 }
719
720 // define the property name list, and put the pk on the head of list.
721 List<String> propertyNames = new ArrayList<>(pks);
722 for (String property : propNames) {
723 if (pks.contains(property)) {
724 continue;
725 }
726 propertyNames.add(property);
727 }
728 return propertyNames;
729 }

739 private List<String> getEdgeProperties(String graphName, String edgeType)
740 throws IOErrorException {
741 String graphType = getGraphType(graphName);
742
743 String descEdgeType = String.format(
744 "CALL describe_graph_type(\"%s\") FILTER type_name=\"%s\" return properties",
745 GqlUtil.escape(graphType), GqlUtil.escape(edgeType));
746 ResultSet resultSet = execute(descEdgeType);
747 if (!resultSet.isSucceeded() || resultSet.isEmpty()) {
748 logger.error(String.format("get description of %s failed for %s", edgeType,
749 resultSet.getErrorMessage()));
750 throw new IllegalArgumentException(String.format("edge type %s does not exist in %s",
751 edgeType, graphName));
752 }
753
754 List<ValueWrapper> properties = resultSet.next().get("properties").asList();
755 return properties.stream().map(ValueWrapper::asString).collect(Collectors.toList());
756 }

765 private String getGraphType(String graphName) throws IOErrorException {
766 ResultSet resultSet = execute(String.format("DESCRIBE GRAPH `%s`",
767 GqlUtil.escape(graphName)));
768 String graphType;
769 if (resultSet.isSucceeded() && !resultSet.isEmpty()) {
770 graphType = resultSet.next().values().get(1).asString();
771 } else {
772 throw new IllegalArgumentException("graphName " + graphName + " does not exist.");
773 }
774 return graphType;
